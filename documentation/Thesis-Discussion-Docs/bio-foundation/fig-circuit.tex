\documentclass[border=8pt]{standalone}

\begin{document}
\begin{tikzpicture}[
  scale=0.8, transform shape,
  box/.style={draw, rectangle, rounded corners, minimum height=1.1cm,
              text width=2.1cm, align=center, font=\scriptsize},
  col/.style={box, fill=blue!5},                 % cortex
  layer/.style={col, text width=3.2cm},          % laminae in the stack
  thn/.style={box, fill=white},                  % thalamic nuclei
  ext/.style={box, fill=black!8},                % external: periphery / other areas
  entry/.style={ext, line width=1.2pt},          % start of the flow
  driver/.style={-{Stealth[length=2.5mm]}, line width=1.2pt},
  intra/.style={-{Stealth[length=2mm]}, line width=0.6pt},
  modulator/.style={-{Stealth[length=2.4mm]}, line width=0.8pt, draw=gray!80},
  modline/.style={line width=0.8pt, draw=gray!80},
  gate/.style={-{Bar[width=3.5mm]}, line width=1.1pt, draw=blue!60!black},
  topdown/.style={-{Stealth[length=2mm]}, line width=0.8pt, dashed},
  bignum/.style={circle, draw, fill=black!80, text=white,
                 font=\scriptsize\bfseries, minimum size=4.5mm, inner sep=0pt},
  region/.style={draw=gray!55, dashed, rounded corners=6pt,
                 fill=gray!6, inner sep=9pt},
  regionlabel/.style={font=\small\bfseries, text=gray!65!black},
  legtext/.style={font=\scriptsize, anchor=west},
]
  % ---- Cortical column (area A), anatomical order, top to bottom ----
  \node[layer] (l1)  at (0,7.2) {\textbf{L1}\\apical tufts};
  \node[layer] (l23) at (0,5.4) {\textbf{L2/3}\\associative};
  \node[layer] (l4)  at (0,3.6) {\textbf{L4}\\granular input};
  \node[layer] (l5)  at (0,1.8) {\textbf{L5}\\deep output};
  \node[layer] (l6)  at (0,0)   {\textbf{L6}\\corticothalamic};

  % ---- Top-down source, above L1 ----
  \node[ext, text width=3cm] (hi) at (0,9.4) {\textbf{Higher cortical area}\\(top-down)};

  % ---- Thalamus: FO relay left, TRN under L6, HO relay right ----
  \node[thn] (fo)  at (-3.6,-3.2) {\textbf{First-order}\\\textbf{relay}\\(LGN, VPM)};
  \node[thn] (trn) at (0,-3.2)    {\textbf{TRN}\\gate};
  \node[thn] (ho)  at (3.6,-3.2)  {\textbf{Higher-order}\\\textbf{relay}\\(pulvinar, Po)};

  % ---- External nodes ----
  \node[entry] (sens) at (-7.6,-3.2) {\textbf{Sensory input}\\(retina, lemniscus)};
  \node[col, dashed] (b) at (7.6,-3.2) {\textbf{L4 of area B}\\(next area)};

  % ---- Regions (drawn behind everything) ----
  \coordinate (ctxW) at (-2.4,3.6);   % widen cortex region to hold the side routes
  \coordinate (ctxE) at (2.7,3.6);
  \begin{scope}[on background layer]
    \node[region, fit=(l1)(l6)(ctxW)(ctxE)] (ctxA) {};
    \node[region, fit=(fo)(trn)(ho)] (thal) {};
  \end{scope}
  \node[regionlabel, anchor=south east] at (ctxA.north east) {Area A (cortex)};
  \node[regionlabel, anchor=north west] at (thal.south west) {Thalamus};

  % ==== TOP-DOWN FEEDBACK (dashed, class open) ====
  \draw[topdown] (hi) -- (l1);                                          % higher area -> L1
  \draw[topdown] ($(l1.south)+(-0.8,0)$) -- ($(l23.north)+(-0.8,0)$);  % L1 tufts -> L2/3
  \draw[topdown] (l1.west) -- (-2.1,7.2) -- (-2.1,1.8) -- (l5.west);    % L1 tufts -> L5 (left side)

  % ==== DIRECT CORTICOCORTICAL FEEDFORWARD (dashed, class open) ====
  \draw[topdown] ($(l23.east)+(0,0.15)$)
        -- node[above, pos=0.6, font=\scriptsize\itshape] {direct corticocortical} (7.6,5.55)
        -- (b.north);                                                   % L2/3 -> L4 of area B

  % ==== ENTRY STUB ====
  \draw[driver] ([xshift=-9mm]sens.west) -- (sens.west);

  % ==== DRIVER PATH (numbered 1-6, thick) ====
  \draw[driver] (sens) -- node[bignum, above=3pt, pos=0.4] {1} (fo);    % sensory -> FO relay
  \draw[driver] (fo.north) |- node[bignum, left=3pt, pos=0.25] {2} (l4.west);  % FO relay -> L4
  \draw[driver] (l4) -- node[bignum, right=3pt] {3} (l23);              % L4 -> L2/3
  \draw[driver] ($(l23.east)+(0,-0.15)$) -- (2.5,5.25)
        -- node[bignum, right=3pt] {4} (2.5,1.95)
        -- ($(l5.east)+(0,0.15)$);                                      % L2/3 -> L5 (around L4)
  \draw[driver] ($(l5.east)+(0,-0.15)$) -- (3.6,1.65)
        -- node[bignum, right=3pt] {5} (ho.north);                      % L5 -> HO relay (past L6)
  \draw[driver] (ho) -- node[bignum, above=3pt, pos=0.62] {6} (b);      % HO relay -> area B

  % ==== Intracortical (not part of the driver path, no badge) ====
  \draw[intra] (l5) -- (l6);                                            % L5 -> L6

  % ==== L6 MODULATORY FEEDBACK (thin grey) ====
  \draw[modline] (l6.south) -- (0,-1.5);
  \draw[modline] (-2.8,-1.5) -- (2.8,-1.5);
  \node[font=\scriptsize\itshape, text=gray!60!black, above] at (-1.6,-1.5) {L6 feedback};
  \draw[modulator] (0,-1.5)    -- (trn.north);                          % L6 -> TRN
  \draw[modulator] (-2.8,-1.5) -- ($(fo.north)+(0.8,0)$);               % L6 -> FO relay
  \draw[modulator] (2.8,-1.5)  -- ($(ho.north)+(-0.8,0)$);              % L6 -> HO relay
  \fill[gray!80] (0,-1.5) circle (1.4pt);

  % ==== TRN GATE (inhibitory, bar-ended) ====
  \draw[gate] (trn.west) -- (fo.east);                                  % TRN -| FO relay
  \draw[gate] (trn.east) -- (ho.west);                                  % TRN -| HO relay

  % ==== LEGEND ====
  \begin{scope}[shift={(-6.25,-5.5)}]
    % row 1
    \draw[driver]    (-1.5,0) -- (-0.6,0);
    \node[legtext] at (-0.4,0) {Driver (information)};
    \draw[modulator] (4.2,0)  -- (5.1,0);
    \node[legtext] at (5.3,0)  {Modulator (gain / state)};
    \draw[gate]      (9.0,0)  -- (9.9,0);
    \node[legtext] at (10.1,0) {TRN inhibition (gate)};
    % row 2
    \draw[topdown]   (-1.5,-0.6) -- (-0.6,-0.6);
    \node[legtext] at (-0.4,-0.6) {Direct corticocortical (class open)};
    \draw[intra]     (4.2,-0.6)  -- (5.1,-0.6);
    \node[legtext] at (5.3,-0.6)  {Intracortical};
    \node[bignum]  at (9.45,-0.6) {1};
    \node[legtext] at (10.1,-0.6) {Signal-path order (1--6)};
    % frame
    \draw[gray!55, rounded corners=4pt] (-1.8,0.35) rectangle (13.4,-0.95);
  \end{scope}
\end{tikzpicture}
